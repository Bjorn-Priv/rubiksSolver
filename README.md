controls: 
    arrow keys move cam
    shift + equals  = zoom in
    shift + minus   = zoom out
    numpad plus     = zoom in
    numpad minus    = zoom out

    f = front
    b = back
    m = middle
    l = left
    r = right
    e = equator
    s = slice
    u = upper
    d = down
    l_ctrl = shuffle
    1 = solve (WIP)
